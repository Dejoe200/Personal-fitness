<script>
/* ---------- Supabase Configuration ---------- */
const SUPABASE_URL = 'https://kqfnrggvprrqrzaqllcb.supabase.co';
const SUPABASE_KEY = 'sb_publishable_glWFOI6tySme3Ik6...'; // Replace with your FULL actual publishable key
const supabaseClient = window.supabase.createClient(SUPABASE_URL, SUPABASE_KEY);

/* ---------- State Management ---------- */
let currentUser = null;
let logs = [];  // Array of 'YYYY-MM-DD' strings
let goals = []; // Array of goal objects
let isLoading = true;

// Listen to Auth State Changes
supabaseClient.auth.onAuthStateChange((event, session) => {
  if (session) {
    currentUser = session.user;
    fetchData();
  } else {
    currentUser = null;
    logs = [];
    goals = [];
    isLoading = false;
    render();
  }
});

/* ---------- Data Layer (Backend API Calls) ---------- */

async function fetchData() {
  isLoading = true;
  render();

  try {
    const [logsRes, goalsRes] = await Promise.all([
      supabaseClient.from('logs').select('logged_date').eq('user_id', currentUser.id),
      supabaseClient.from('goals').select('*').eq('user_id', currentUser.id).order('created_at', { ascending: true })
    ]);

    if (logsRes.data) logs = logsRes.data.map(l => l.logged_date);
    if (goalsRes.data) goals = goalsRes.data;
  } catch (err) {
    showToast('Failed to load data', false);
  }

  isLoading = false;
  render();
}

async function handleAuth(action) {
  const email = document.getElementById('email-input').value.trim();
  const password = document.getElementById('password-input').value;

  if (!email || !password) {
    showToast('Email and password required', false);
    return;
  }

  isLoading = true;
  render();

  let res;
  if (action === 'login') {
    res = await supabaseClient.auth.signInWithPassword({ email, password });
  } else {
    res = await supabaseClient.auth.signUp({ email, password });
  }

  if (res.error) {
    showToast(res.error.message, false);
    isLoading = false;
    render();
  } else {
    showToast(action === 'login' ? 'Welcome back!' : 'Sign up successful!');
  }
}

async function handleLogout() {
  await supabaseClient.auth.signOut();
}

async function toggleToday() {
  const t = todayStr();
  const isLogged = isLoggedOn(t);

  // Optimistic UI update
  if (isLogged) {
    logs = logs.filter(d => d !== t);
  } else {
    logs.push(t);
  }
  render();

  // API sync
  let error;
  if (isLogged) {
    const res = await supabaseClient.from('logs').delete().match({ user_id: currentUser.id, logged_date: t });
    error = res.error;
  } else {
    const res = await supabaseClient.from('logs').insert([{ user_id: currentUser.id, logged_date: t }]);
    error = res.error;
  }

  if (error) {
    // Revert optimistic update on failure
    if (isLogged) logs.push(t);
    else logs = logs.filter(d => d !== t);
    render();
    showToast('Failed to save to cloud', false);
  } else {
    showToast(isLogged ? 'Unmarked for today' : 'Workout logged for today');
  }
}

async function addGoal() {
  const textInput = document.getElementById('new-goal-text');
  const dateInput = document.getElementById('new-goal-date');
  const text = textInput.value.trim();
  const target = dateInput.value;

  if (!text) return;

  const newGoal = {
    user_id: currentUser.id,
    goal_text: text,
    target_date: target || null,
    is_completed: false
  };

  const { data, error } = await supabaseClient.from('goals').insert([newGoal]).select();

  if (error) {
    showToast('Failed to add goal', false);
  } else if (data) {
    goals.push(data[0]);
    showToast('Goal added');
    render();
  }
}

async function toggleGoal(id, currentStatus) {
  const newStatus = !currentStatus;
  
  // Optimistic UI update
  const g = goals.find(g => g.id === id);
  if (g) g.is_completed = newStatus;
  render();

  const { error } = await supabaseClient.from('goals').update({ is_completed: newStatus }).eq('id', id);
  
  if (error) {
    if (g) g.is_completed = currentStatus; // Revert on failure
    render();
    showToast('Failed to update goal', false);
  }
}

async function deleteGoal(id) {
  const { error } = await supabaseClient.from('goals').delete().eq('id', id);
  if (!error) {
    goals = goals.filter(g => g.id !== id);
    showToast('Goal deleted');
    render();
  } else {
    showToast('Failed to delete goal', false);
  }
}

/* ---------- Utility Functions ---------- */

function todayStr(offsetDays = 0) {
  const d = new Date();
  d.setDate(d.getDate() + offsetDays);
  return d.toISOString().slice(0,10);
}

function isLoggedOn(dateStr) { return logs.includes(dateStr); }

function computeStreak() {
  const logSet = new Set(logs);
  let streak = 0;
  let cursor = logSet.has(todayStr()) ? 0 : -1; 
  if (cursor === -1 && !logSet.has(todayStr(-1))) return 0; 
  let offset = logSet.has(todayStr()) ? 0 : -1;
  while (logSet.has(todayStr(offset))) {
    streak++;
    offset--;
  }
  return streak;
}

function totalSessions() { return logs.length; }

function weekDays() {
  const days = [];
  for (let i = -6; i <= 0; i++) {
    const d = new Date();
    d.setDate(d.getDate() + i);
    days.push({
      label: d.toLocaleDateString('en-US', { weekday: 'narrow' }),
      dateStr: d.toISOString().slice(0,10),
      isToday: i === 0
    });
  }
  return days;
}

function showToast(msg, positive = true) {
  const toast = document.getElementById('toast');
  const text = document.getElementById('toast-text');
  text.textContent = msg;
  toast.querySelector('i').className = positive ? 'fa-solid fa-check text-emerald-400' : 'fa-solid fa-xmark text-zinc-400';
  toast.classList.remove('hidden');
  toast.style.transform = 'translate(-50%, 0)';
  clearTimeout(window._toastTimer);
  window._toastTimer = setTimeout(() => {
    toast.style.transform = 'translate(-50%, 100px)';
    setTimeout(() => toast.classList.add('hidden'), 500);
  }, 2200);
}

function escapeHTML(str) {
  const div = document.createElement('div');
  div.textContent = str;
  return div.innerHTML;
}

    app.innerHTML = renderAuth();
  }
